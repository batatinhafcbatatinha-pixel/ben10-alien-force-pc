set(PS2X_REQUIRE_GPU_RASTERIZER ${BEN10_REQUIRE_GPU_RASTERIZER} CACHE BOOL "Ben10 requires native host GPU rasterization" FORCE)
set(PS2X_GPU_COMPUTE_SCANOUT ON CACHE BOOL "GPU detile, CRT blend and field scanout" FORCE)

if(BEN10_PREBUILT_AOT AND NOT EXISTS "${BEN10_PREBUILT_AOT}")
    message(FATAL_ERROR "BEN10_PREBUILT_AOT does not exist")
endif()
set(PS2X_GAME_LIB_PREBUILT "${BEN10_PREBUILT_AOT}" CACHE FILEPATH "Link only this game's generated EE code" FORCE)
if(WIN32)
    set(PS2X_FFMPEG_ROOT "${WORK_ROOT}/codwaw-ff/build/game/ThirdParty/ffmpeg-prefix/src/ffmpeg_external" CACHE PATH "Reuse installed decoder, no duplicate download")

    set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Never download build dependencies" FORCE)

    if(NOT EXISTS "${FETCHCONTENT_SOURCE_DIR_RAYLIB}/CMakeLists.txt")
        message(FATAL_ERROR "Existing raylib checkout required; network acquisition is disabled")
    endif()
else()
    # Linux/other POSIX: FFmpeg comes from the system (pkg-config); raylib 5.5 is fetched once unless
    # FETCHCONTENT_SOURCE_DIR_RAYLIB points at an existing checkout (see linux/README-linux.md).
endif()
# PS2Recomp runtime: the third_party/PS2Recomp submodule (fork, branch native-ports) when checked out,
# otherwise the shared checkout next to the game folders (../third_party/PS2Recomp).
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/PS2Recomp/CMakeLists.txt")
    set(BEN10_PS2RECOMP_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/PS2Recomp")
else()
    set(BEN10_PS2RECOMP_DIR "${WORK_ROOT}/third_party/PS2Recomp")
endif()
get_filename_component(BEN10_PS2RECOMP_DIR "${BEN10_PS2RECOMP_DIR}" ABSOLUTE)
add_subdirectory("${BEN10_PS2RECOMP_DIR}" ps2recomp)
if(MSVC)
    # Debug info for the runtime and entry (not the AOT game library): cdb stacks with names. /Z7 works with sccache.
    target_compile_options(ps2_runtime PRIVATE /Z7)
    target_compile_options(ps2EntryRunner PRIVATE /Z7)
    target_link_options(ps2EntryRunner PRIVATE /DEBUG:FULL)
else()
    # Symbols for gdb/perf stacks on the runtime and entry (not on the large AOT library).
    target_compile_options(ps2_runtime PRIVATE -g1)
    target_compile_options(ps2EntryRunner PRIVATE -g1)
endif()
get_target_property(BEN10_ENTRY_SOURCES ps2EntryRunner SOURCES)
list(REMOVE_ITEM BEN10_ENTRY_SOURCES "${BEN10_PS2RECOMP_DIR}/ps2xRuntime/src/main.cpp")
set_property(TARGET ps2EntryRunner PROPERTY SOURCES ${BEN10_ENTRY_SOURCES})
target_sources(ps2EntryRunner PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp")
target_compile_definitions(ps2EntryRunner PRIVATE "PS2X_SHARED_MAIN_CPP=\"${BEN10_PS2RECOMP_DIR}/ps2xRuntime/src/main.cpp\"")
# Functions outside the Ghidra map (scripts/gen_gap_functions.py); installed into empty table slots by main.cpp.
# Compiled in batches of 64 per translation unit: 1,174 separate /Od compiles ran out of memory (2026-10-26).
file(GLOB BEN10_GAP_FILES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/gap-functions/gap_*.cpp")
list(SORT BEN10_GAP_FILES)
set(BEN10_GAP_SOURCES "")
set(batch "")
set(batch_index 0)
list(LENGTH BEN10_GAP_FILES gap_count)
set(gap_seen 0)
foreach(gap_file IN LISTS BEN10_GAP_FILES)
    string(APPEND batch "#include \"${gap_file}\"
")
    math(EXPR gap_seen "${gap_seen} + 1")
    math(EXPR batch_fill "${gap_seen} % 64")
    if(batch_fill EQUAL 0 OR gap_seen EQUAL gap_count)
        set(unit "${CMAKE_CURRENT_BINARY_DIR}/gap-units/gap_unit_${batch_index}.cpp")
        file(CONFIGURE OUTPUT "${unit}" CONTENT "${batch}" @ONLY)
        list(APPEND BEN10_GAP_SOURCES "${unit}")
        set(batch "")
        math(EXPR batch_index "${batch_index} + 1")
    endif()
endforeach()
target_sources(ps2EntryRunner PRIVATE ${BEN10_GAP_SOURCES})
if(MSVC)
    set_source_files_properties(${BEN10_GAP_SOURCES} PROPERTIES COMPILE_OPTIONS "/Od;/Ob0;/fp:precise;/bigobj")
else()
    set_source_files_properties(${BEN10_GAP_SOURCES} TARGET_DIRECTORY ps2EntryRunner
        PROPERTIES COMPILE_OPTIONS "-O0;-ffp-contract=off")
endif()

# Large resumable switches make whole-game optimizing compiles expensive.
# Bootstrap native code without optimizer expansion; optimize after coverage
# and dispatch integration stabilize. Other projects/runtime retain Release.
option(BEN10_OPTIMIZE_AOT "Optimize the large generated game-code library" OFF)
if(MSVC)
    # Unoptimized AOT emits large native frames; the observed boot exhausted 1 MiB.
    target_link_options(ps2EntryRunner PRIVATE /STACK:16777216 /MAP)
    if(BEN10_OPTIMIZE_AOT)
        target_compile_options(ps2_game_code PRIVATE /O1 /Ob1 /fp:precise)
    else()
        target_compile_options(ps2_game_code PRIVATE /Od /Ob0 /fp:precise)
    endif()
else()
    # GCC/Clang: same policy as MSVC. -ffp-contract=off keeps FMA contraction out (the PS2 FPU has no fused
    # multiply-add); the game thread stack is raised at runtime (PS2X_THREAD_STACK_MB, default 64 MiB).
    if(BEN10_OPTIMIZE_AOT)
        target_compile_options(ps2_game_code PRIVATE -O1 -ffp-contract=off)
    else()
        target_compile_options(ps2_game_code PRIVATE -O0 -ffp-contract=off)
    endif()
endif()
